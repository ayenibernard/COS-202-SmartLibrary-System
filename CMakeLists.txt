cmake_minimum_required(VERSION 3.16)
project(SmartLibraryCirculation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets)
add_executable(SmartLibrary main.cpp)
target_include_directories(SmartLibrary PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(SmartLibrary PRIVATE Qt6::Widgets)
enable_testing()
add_executable(CirculationTests tests/circulation_tests.cpp)
target_include_directories(CirculationTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME CirculationTests COMMAND CirculationTests)
add_executable(SearchTests tests/search_tests.cpp)
target_include_directories(SearchTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME SearchTests COMMAND SearchTests)

add_executable(AdminTests tests/admin_tests.cpp)
target_include_directories(AdminTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME AdminTests COMMAND AdminTests)

add_executable(PersistenceTests tests/persistence_tests.cpp)
target_include_directories(PersistenceTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(PersistenceTests PRIVATE Qt6::Core)
add_test(NAME PersistenceTests COMMAND PersistenceTests)

add_executable(ValidationTests tests/validation_tests.cpp)
target_include_directories(ValidationTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME ValidationTests COMMAND ValidationTests)

add_executable(CacheTests tests/cache_tests.cpp)
target_include_directories(CacheTests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
add_test(NAME CacheTests COMMAND CacheTests)
